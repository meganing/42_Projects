function get_next_line(fd):
    static char *stash;     // save remainder from previous call
    char buff[BUFFER_SIZE + 1];
    char *newline_pos;
    char *line; (for writing out the output of the line read)
    int bytes_read; (the bytes that was read from the read function)

    // Keep reading until we find a newline or reach EOF
    while (newline not found in stash):
        bytes_read = read(fd, buff, BUFFER_SIZE)
        if (bytes_read < 0)
            return NULL
        if (bytes_read == 0)
            break  // EOF reached

        buff[bytes_read] = '\0'
        stash = join(stash, buff)  // concatenate and free old stash
    end while

    // Extract one line from stash (up to \n or end)
    line = extract_line(stash)

    // Update stash to keep what's left after that line
    stash = keep_after_line(stash)

    return line
